# add_pd_external(<target> <output name> <sources...>) : one Pd external with the right suffix per platform
function(add_pd_external target outname)
  add_library(${target} MODULE ${ARGN})
  target_include_directories(${target} PRIVATE "${PD_INCLUDE_DIR}")
  target_compile_options(${target} PRIVATE -O3 -Wall)
  set_target_properties(${target} PROPERTIES OUTPUT_NAME "${outname}" PREFIX "")
  if(APPLE)
    set_target_properties(${target} PROPERTIES SUFFIX ".pd_darwin")
    target_link_options(${target} PRIVATE -undefined dynamic_lookup)
  elseif(WIN32)
    set_target_properties(${target} PROPERTIES SUFFIX ".dll")
    if(PD_LIB)
      target_link_libraries(${target} PRIVATE "${PD_LIB}")
    endif()
  else()
    set_target_properties(${target} PROPERTIES SUFFIX ".pd_linux")
  endif()
  if(NOT WIN32)
    target_link_libraries(${target} PRIVATE m)
  endif()
  install(TARGETS ${target} LIBRARY DESTINATION ${DEST} RUNTIME DESTINATION ${DEST})
endfunction()

# ---------------------------------------------------------------- fx
set(FX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/fx")
file(GLOB fx_hdrs "${FX_DIR}/src/*.h")
add_pd_external(tspec_tilde  "tspec~"  "${FX_DIR}/src/tspec~.c" "${FX_DIR}/src/tspec_core.c" ${fx_hdrs})
add_pd_external(tgrain_tilde "tgrain~" "${FX_DIR}/src/tgrain~.c" ${fx_hdrs})
add_pd_external(tmodal_tilde "tmodal~" "${FX_DIR}/src/tmodal~.c" ${fx_hdrs})
add_pd_external(tdata_tilde  "tdata~"  "${FX_DIR}/src/tdata~.c" ${fx_hdrs})
foreach(t tspec_tilde tgrain_tilde tmodal_tilde tdata_tilde)
  target_include_directories(${t} PRIVATE "${FX_DIR}/src")
endforeach()
file(GLOB fx_abs "${FX_DIR}/abs/*.pd")
file(GLOB fx_help "${FX_DIR}/help/*.pd")
install(FILES ${fx_abs} ${fx_help} DESTINATION ${DEST})

# ---------------------------------------------------------------- synths
# One folder per synth. A synth is built when synths/<name>/<name>~.c exists.
set(SY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/synths")
set(ALL_SYNTHS dx7 sh101 alphajuno b3 prophet5 minimoog tx81z ms20 cz101 mellotron)
foreach(syn ${ALL_SYNTHS})
  set(dir "${SY_DIR}/${syn}")
  if(EXISTS "${dir}/${syn}~.c")
    file(GLOB hdrs "${dir}/*.h" "${SY_DIR}/common/*.h")
    add_pd_external(${syn}_tilde "${syn}~" "${dir}/${syn}~.c" ${hdrs})
    target_include_directories(${syn}_tilde PRIVATE "${dir}" "${SY_DIR}/common")
    file(GLOB pds "${dir}/*.pd")
    file(GLOB banks "${dir}/*.syx")
    install(FILES ${pds} ${banks} DESTINATION ${DEST})
  endif()
endforeach()
